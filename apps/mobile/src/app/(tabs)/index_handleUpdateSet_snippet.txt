// Workout handlers with multipliers
  const handleUpdateSet = (exId, setIdx, field, val) => {
    const next = workout.map((ex) =>
      ex.id !== exId
        ? ex
        : {
            ...ex,
            sets: ex.sets.map((s, i) =>
              i !== setIdx ? s : { ...s, [field]: val },
            ),
          },
    );
    setWorkout(next);
    
    // Calculate volume with multipliers
    let total = 0;
    next.forEach((ex) =>
      ex.sets.forEach((s) => {
        const weight = parseFloat(s.weight) || 0;
        const reps = parseFloat(s.reps) || 0;
        const baseVolume = weight * reps;
        
        // Apply equipment multiplier
        const equipmentMultipliers = {
          dumbbell: 1.15,
          barbell: 1.0,
          cable: 0.85,
        };
        const equipmentMult = equipmentMultipliers[s.equipment] || 1.0;
        
        // Apply rep quality multiplier (default 1.0)
        const qualityMult = parseFloat(s.rep_quality) || 1.0;
        
        // Calculate modified volume (without Wilks, backend will add that)
        const modifiedVolume = baseVolume * equipmentMult * qualityMult;
        
        total += modifiedVolume;
      }),
    );
    setVolume(total);
  };

  const handleAddSet = (exId) =>
    setWorkout((w) =>
      w.map((ex) =>
        ex.id !== exId
          ? ex
          : { 
              ...ex, 
              sets: [...ex.sets, { weight: "", reps: "", equipment: "barbell", rep_quality: 1.0 }] 
            },
      ),
    );
